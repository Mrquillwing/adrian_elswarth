    <script>
        document.getElementById('download-btn').addEventListener('click', () => {
            const banner = document.getElementById('github-banner');
            let cssStyles = '';
            for (let sheet of document.styleSheets) {
                try {
                    for (let rule of sheet.cssRules) {
                        cssStyles += rule.cssText + '\n';
                    }
                } catch(e) {}
            }

            const svg = `
                <svg xmlns="http://www.w3.org/2000/svg" width="1200" height="420">
                    <defs><style>${cssStyles}</style></defs>
                    <foreignObject width="1200" height="420">
                        <div xmlns="http://www.w3.org/1999/xhtml">${banner.outerHTML}</div>
                    </foreignObject>
                </svg>
            `;

            const blob = new Blob([svg], { type: 'image/svg+xml;charset=utf-8' });
            const url = URL.createObjectURL(blob);
            const img = new Image();
            
            img.onload = () => {
                const canvas = document.createElement('canvas');
                canvas.width = 1200;
                canvas.height = 420;
                const ctx = canvas.getContext('2d');
                ctx.fillStyle = '#06110D';
                ctx.fillRect(0, 0, 1200, 420);
                ctx.drawImage(img, 0, 0);
                
                const a = document.createElement('a');
                a.href = canvas.toDataURL('image/png');
                a.download = 'ai-neural-banner.png';
                a.click();
                URL.revokeObjectURL(url);
            };
            img.src = url;
        });
    </script>
